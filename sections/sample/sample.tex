\textbf{bra-ket notation}
\begin{gather}
    |0\rangle =
    \begin{pmatrix}\label{ket0}
        1 \\
        0
    \end{pmatrix} \\
    |1\rangle =
    \begin{pmatrix}\label{ket1}
        0 \\
        1
    \end{pmatrix}
\end{gather}

\textbf{Outer product}
\begin{equation}
    |0\rangle\langle 1| =
    \begin{bmatrix}
        1 \\
        0
    \end{bmatrix}
    \begin{bmatrix}
        0 & 1
    \end{bmatrix} =
    \begin{bmatrix}
        1 \cdot 0 & 1 \cdot 1 \\
        0 \cdot 0 & 0 \cdot 1
    \end{bmatrix} =
    \begin{bmatrix}
        0 & 1 \\
        0 & 0
    \end{bmatrix}
    \notag
\end{equation}

\textbf{tensor product}
\begin{equation}
    \begin{pmatrix}
        a \\
        b
    \end{pmatrix}
    \otimes
    \begin{pmatrix}
        c \\
        d
    \end{pmatrix} =
    \begin{pmatrix}
        a \ast c \\
        a \ast d \\
        b \ast c \\
        b \ast d
    \end{pmatrix}
\end{equation}

\textbf{qubit representation}
\begin{equation}\label{qubit-eq}
    |\psi\rangle = a|0\rangle + b|1\rangle, \quad a,b \in \mathbb{C}, \quad
    |a|^{2} + |b|^{2} = 1, \quad |\psi\rangle \sim e^{i\phi}|\psi\rangle.
\end{equation}
